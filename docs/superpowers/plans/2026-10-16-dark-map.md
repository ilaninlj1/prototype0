# The Dark Map Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every measured song on one map placed by how it sounds; heard songs light up, the dark is your blind spot, and tapping a dark spot makes Home play unheard songs from there, blind.

**Architecture:** A dev-only script lays the sound index out with UMAP and writes `assets/sound-map.json` (coordinates, regions with plain-word names, genre per song). Pure modules (`lib/sound-map.ts`, `lib/dark-map.ts`) place songs, light the map and pick songs near a spot; `lib/dark-map-app.ts` feeds them from storage. A full-screen Skia map (`app/map.tsx`) sends a steering request; Home serves `getTracksNear` while a `mapFocus` is set.

**Tech Stack:** Expo SDK 57, Expo Router 6, React Native, `@shopify/react-native-skia` 2.6.2, Reanimated 4.5, Gesture Handler, AsyncStorage; `umap-js` 1.4.0 (dev only); Node's test runner with `--experimental-strip-types`.

// lib/sound-map.ts
// The Dark Map's data: every measured song placed on one map by how it sounds (assets/sound-map.json, built by
// scripts/build-sound-map.ts), and the pure helpers that place other songs, look around a spot and pick songs
// near it. The map runs 0–1 across (x) and 0–1 down (y).

import type { SoundFeatures } from './sound.ts';

/** The measures the map is laid out from, in this order everywhere: the file, the vectors, the words. */
export const MAP_MEASURES = ['energy', 'danceability', 'acousticness', 'instrumentalness', 'speechiness', 'valence', 'loudness', 'tempo'] as const;
export type MapMeasure = (typeof MAP_MEASURES)[number];

export type MapPoint = { x: number; y: number };
export type MapRegion = { x: number; y: number; words: string; songs: number };
/** x, y, region index, genre index */
export type MapSong = [number, number, number, number];
export type SoundMap = {
  v: 1;
  builtAt: string;
  settings: { neighbors: number; minDist: number; seed: number; regions: number };
  measures: string[];
  mean: number[];
  sd: number[];
  genres: string[];
  regions: MapRegion[];
  songs: Record<string, MapSong>;
};
/** A map song with its standardized measures, for placing songs that aren't on the map. */
export type Neighbor = { x: number; y: number; region: number; vector: number[] };

let cached: SoundMap | null | undefined;

/** The bundled map, read on first use. Plain Node (tests, scripts) has no `require`, so it's null there. */
export function loadSoundMap(): SoundMap | null {
  if (cached === undefined) cached = typeof require === 'function' ? (require('../assets/sound-map.json') as SoundMap) : null;
  return cached;
}

/** A song's eight measures as numbers, NaN where one is missing. */
export function measureVector(f: Partial<Record<MapMeasure, number | null>>): number[] {
  return MAP_MEASURES.map((m) => {
    const v = f[m];
    return typeof v === 'number' && Number.isFinite(v) ? v : Number.NaN;
  });
}

/** z-scores against the map's means and spreads; a missing measure sits at the mean. */
export function standardize(map: Pick<SoundMap, 'mean' | 'sd'>, raw: number[]): number[] {
  return raw.map((v, j) => (Number.isFinite(v) ? (v - map.mean[j]) / (map.sd[j] || 1) : 0));
}

const d2 = (a: number[], b: number[]) => a.reduce((sum, v, j) => sum + (v - b[j]) ** 2, 0);
const gap = (song: MapSong, p: MapPoint) => Math.hypot(song[0] - p.x, song[1] - p.y);

/** Every map song with its standardized measures; songs `featuresOf` can't measure are left out. */
export function mapNeighbors(map: SoundMap, featuresOf: (id: number) => SoundFeatures | null): Neighbor[] {
  const out: Neighbor[] = [];
  for (const [id, [x, y, region]] of Object.entries(map.songs)) {
    const f = featuresOf(Number(id));
    if (f) out.push({ x, y, region, vector: standardize(map, measureVector(f)) });
  }
  return out;
}

/** A song that isn't on the map: the average spot of its k nearest map songs by sound, in the region most of them share. */
export function placeByNeighbors(vector: number[], neighbors: Neighbor[], k = 5): { x: number; y: number; region: number } | null {
  if (neighbors.length === 0) return null;
  const nearest = neighbors
    .map((n) => ({ n, d: d2(vector, n.vector) }))
    .sort((a, b) => a.d - b.d)
    .slice(0, k)
    .map((e) => e.n);
  const votes = new Map<number, number>();
  for (const n of nearest) votes.set(n.region, (votes.get(n.region) ?? 0) + 1);
  // Most votes wins; a tie goes to the region of the nearest song.
  let region = nearest[0].region;
  for (const [r, v] of votes) if (v > (votes.get(region) ?? 0)) region = r;
  return {
    x: nearest.reduce((s, n) => s + n.x, 0) / nearest.length,
    y: nearest.reduce((s, n) => s + n.y, 0) / nearest.length,
    region,
  };
}

/** Ids of the map songs within `radius` of a point. */
export function songsWithin(map: Pick<SoundMap, 'songs'>, point: MapPoint, radius: number): number[] {
  return Object.entries(map.songs)
    .filter(([, s]) => gap(s, point) <= radius)
    .map(([id]) => Number(id));
}

/** The region most songs around a point belong to; on open sea, the region with the nearest center. */
export function regionAt(map: Pick<SoundMap, 'songs' | 'regions'>, point: MapPoint, radius: number): number {
  const votes = new Map<number, number>();
  for (const s of Object.values(map.songs)) if (gap(s, point) <= radius) votes.set(s[2], (votes.get(s[2]) ?? 0) + 1);
  if (votes.size > 0) return [...votes].sort((a, b) => b[1] - a[1] || a[0] - b[0])[0][0];
  const dist = (r: MapRegion) => Math.hypot(r.x - point.x, r.y - point.y);
  let best = 0;
  map.regions.forEach((r, i) => {
    if (dist(r) < dist(map.regions[best])) best = i;
  });
  return best;
}

/** Fisher–Yates with an injectable random, so tests can keep the order. */
function shuffle<T>(list: T[], random: () => number): T[] {
  for (let i = list.length - 1; i > 0; i--) {
    const j = Math.floor(random() * (i + 1));
    [list[i], list[j]] = [list[j], list[i]];
  }
  return list;
}

/**
 * Songs to play from a spot: the unheard map songs nearest it that `resolve` can turn into a track and `accept`
 * lets through, one per artist. The nearest 3 × n are shuffled and n kept, so a spot plays varied songs; since the
 * nearest are taken first, the search widens by itself as a spot runs dry. Empty when nothing is left.
 */
export function pickNear<T>(
  map: Pick<SoundMap, 'songs' | 'genres'>,
  point: MapPoint,
  n: number,
  heard: Set<number>,
  resolve: (id: number, genre: string) => { artist: string; track: T } | null,
  accept: (artist: string) => boolean,
  random: () => number = Math.random
): T[] {
  const ids = Object.keys(map.songs)
    .map(Number)
    .filter((id) => !heard.has(id));
  ids.sort((a, b) => gap(map.songs[a], point) - gap(map.songs[b], point));
  const found: T[] = [];
  const artists = new Set<string>();
  for (const id of ids) {
    if (found.length >= 3 * n) break;
    const r = resolve(id, map.genres[map.songs[id][3]]);
    if (!r || artists.has(r.artist) || !accept(r.artist)) continue;
    artists.add(r.artist);
    found.push(r.track);
  }
  return shuffle(found, random).slice(0, n);
}

// scripts/sound-map-rules.ts
// The pure parts of building the Dark Map (scripts/build-sound-map.ts): standardizing the measures, cutting the
// layout into regions and naming each region in plain words.

import { MAP_MEASURES, type MapMeasure } from '../lib/sound-map.ts';

type Pt = [number, number];

/** A seeded random in [0, 1), so a rebuild of the same index gives the same map. */
export function seeded(seed: number): () => number {
  let s = seed % 2147483647 || 1;
  return () => (s = (s * 16807) % 2147483647) / 2147483647;
}

/** Each column's missing values (NaN) filled with its mean, then turned into z-scores. */
export function standardizeColumns(raw: number[][]): { rows: number[][]; mean: number[]; sd: number[] } {
  const cols = raw[0]?.length ?? 0;
  const mean = Array.from({ length: cols }, (_, j) => {
    const known = raw.map((r) => r[j]).filter((v) => Number.isFinite(v));
    return known.length ? known.reduce((a, b) => a + b, 0) / known.length : 0;
  });
  const filled = raw.map((r) => r.map((v, j) => (Number.isFinite(v) ? v : mean[j])));
  const sd = mean.map((m, j) => Math.sqrt(filled.reduce((a, r) => a + (r[j] - m) ** 2, 0) / Math.max(1, filled.length)) || 1);
  return { rows: filled.map((r) => r.map((v, j) => (v - mean[j]) / sd[j])), mean, sd };
}

/** Scales a layout so both axes run from 0 to 1. */
export function normalizeLayout(points: number[][]): Pt[] {
  const xs = points.map((p) => p[0]);
  const ys = points.map((p) => p[1]);
  const lo = [Math.min(...xs), Math.min(...ys)];
  const hi = [Math.max(...xs), Math.max(...ys)];
  return points.map((p) => [(p[0] - lo[0]) / (hi[0] - lo[0] || 1), (p[1] - lo[1]) / (hi[1] - lo[1] || 1)]);
}

const d2 = (a: Pt, b: Pt) => (a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2;

function nearestCenter(p: Pt, centers: Pt[], skip = -1): number {
  let best = -1;
  centers.forEach((c, i) => {
    if (i !== skip && (best < 0 || d2(p, c) < d2(p, centers[best]))) best = i;
  });
  return best;
}

function centersOf(points: Pt[], assign: number[], k: number): Pt[] {
  return Array.from({ length: k }, (_, c) => {
    const members = points.filter((_, i) => assign[i] === c);
    return [members.reduce((s, p) => s + p[0], 0) / members.length, members.reduce((s, p) => s + p[1], 0) / members.length];
  });
}

/** k-means on the layout: k-means++ seeding from `random`, then Lloyd rounds. */
export function kmeans(points: Pt[], k: number, random: () => number, rounds = 40): { centers: Pt[]; assign: number[] } {
  const centers: Pt[] = [points[Math.floor(random() * points.length)]];
  while (centers.length < k) {
    const far = points.map((p) => Math.min(...centers.map((c) => d2(p, c))));
    let pick = random() * far.reduce((a, b) => a + b, 0);
    let i = 0;
    while (i < points.length - 1 && (pick -= far[i]) > 0) i++;
    centers.push(points[i]);
  }
  let assign: number[] = [];
  for (let round = 0; round < rounds; round++) {
    assign = points.map((p) => nearestCenter(p, centers));
    centers.forEach((c, ci) => {
      const members = points.filter((_, i) => assign[i] === ci);
      if (members.length) centers[ci] = [members.reduce((s, p) => s + p[0], 0) / members.length, members.reduce((s, p) => s + p[1], 0) / members.length];
    });
  }
  return { centers, assign };
}

/** Regions under `min` points join the region with the nearest center; then they're renumbered by size, largest first. */
export function mergeSmall(points: Pt[], assign: number[], min: number): { centers: Pt[]; assign: number[] } {
  let next = [...assign];
  for (;;) {
    const ids = [...new Set(next)];
    const counts = new Map(ids.map((r) => [r, next.filter((a) => a === r).length]));
    const small = ids.filter((r) => counts.get(r)! < min).sort((a, b) => counts.get(a)! - counts.get(b)! || a - b)[0];
    if (small === undefined || ids.length < 2) break;
    const centers = new Map(ids.map((r) => [r, centersOf(points, next.map((a) => (a === r ? 0 : -1)), 1)[0]]));
    const target = ids
      .filter((r) => r !== small)
      .sort((a, b) => d2(centers.get(small)!, centers.get(a)!) - d2(centers.get(small)!, centers.get(b)!) || a - b)[0];
    next = next.map((a) => (a === small ? target : a));
  }
  const bySize = [...new Set(next)].sort((a, b) => next.filter((x) => x === b).length - next.filter((x) => x === a).length || a - b);
  const renumbered = next.map((a) => bySize.indexOf(a));
  return { centers: centersOf(points, renumbered, bySize.length), assign: renumbered };
}

const WORDS: Record<MapMeasure, [low: string, high: string]> = {
  energy: ['calm', 'loud'],
  danceability: ['loose', 'groovy'],
  acousticness: ['electric', 'acoustic'],
  instrumentalness: ['sung', 'instrumental'],
  speechiness: ['melodic', 'wordy'],
  valence: ['dark', 'bright'],
  loudness: ['soft', 'loud'],
  tempo: ['slow', 'fast'],
};

/** The measures a region differs on most, strongest first. Loudness is left out: it repeats energy. */
function ranked(meanZ: number[]): string[] {
  return MAP_MEASURES.map((measure, j) => ({ measure, z: meanZ[j] }))
    .filter((m) => m.measure !== 'loudness')
    .sort((a, b) => Math.abs(b.z) - Math.abs(a.z))
    .map((m) => WORDS[m.measure][m.z > 0 ? 1 : 0]);
}

/** Two plain words per region ("instrumental · groovy"), three when two regions would share the same two. */
export function nameRegions(meanZs: number[][]): string[] {
  const two = meanZs.map((z) => ranked(z).slice(0, 2).join(' · '));
  return meanZs.map((z, i) => (two.filter((t) => t === two[i]).length > 1 ? ranked(z).slice(0, 3).join(' · ') : two[i]));
}

// scripts/build-sound-map.ts
// Builds assets/sound-map.json, the Dark Map: every song in assets/sound-index.json laid out by how it sounds
// (UMAP over eight standardized measures, seeded, so the same index always gives the same map), cut into regions
// with plain-word names. Dev only: umap-js is a devDependency the app never imports.
//
//   npm run build-sound-map

import fs from 'node:fs';
import path from 'node:path';
import { UMAP } from 'umap-js';

import { fromIndexRow } from '../lib/sound.ts';
import { MAP_MEASURES, measureVector, type MapSong, type SoundMap } from '../lib/sound-map.ts';
import { kmeans, mergeSmall, nameRegions, normalizeLayout, seeded, standardizeColumns } from './sound-map-rules.ts';

const ROOT = path.resolve(import.meta.dirname, '..');
const SETTINGS = { neighbors: 15, minDist: 0.02, seed: 42, regions: 8 };
const MIN_REGION = 60;
const r3 = (v: number) => Math.round(v * 1000) / 1000;
const r4 = (v: number) => Math.round(v * 10000) / 10000;

type Entry = { itunesTrackId: number };
type Catalog = Record<string, { hits: Entry[]; deepCuts: Entry[]; mixed?: Entry[] }>;

// The same rule scripts/precompute-catalogs.ts names catalog files by.
function slugifyGenre(genre: string): string {
  return genre
    .toLowerCase()
    .normalize('NFKD')
    .replace(/[̀-ͯ]/g, '')
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '');
}

/** Each catalog song's genre tag (the name lib/pool.ts loadCatalog uses): the first genre, alphabetically, that lists it. */
function genreTags(): Map<number, string> {
  const tags = Object.keys(JSON.parse(fs.readFileSync(path.join(ROOT, 'assets/genres.json'), 'utf8'))).sort();
  const out = new Map<number, string>();
  for (const tag of tags) {
    const file = path.join(ROOT, 'assets/catalogs', `${slugifyGenre(tag)}.json`);
    if (!fs.existsSync(file)) {
      console.warn(`no catalog for ${tag}`);
      continue;
    }
    const catalog = JSON.parse(fs.readFileSync(file, 'utf8')) as Catalog;
    for (const a of Object.values(catalog))
      for (const e of [...a.hits, ...a.deepCuts, ...(a.mixed ?? [])]) if (!out.has(e.itunesTrackId)) out.set(e.itunesTrackId, tag);
  }
  return out;
}

function main() {
  const index = JSON.parse(fs.readFileSync(path.join(ROOT, 'assets/sound-index.json'), 'utf8')) as { songs: Record<string, unknown> };
  const tagOf = genreTags();
  const ids = Object.keys(index.songs)
    .map(Number)
    .filter((id) => tagOf.has(id))
    .sort((a, b) => a - b);
  const { rows, mean, sd } = standardizeColumns(ids.map((id) => measureVector(fromIndexRow(index.songs[String(id)]) ?? {})));

  const random = seeded(SETTINGS.seed);
  console.time('layout');
  const layout = new UMAP({ nComponents: 2, nNeighbors: SETTINGS.neighbors, minDist: SETTINGS.minDist, random }).fit(rows);
  console.timeEnd('layout');
  const points = normalizeLayout(layout);
  const { centers, assign } = mergeSmall(points, kmeans(points, SETTINGS.regions, random).assign, MIN_REGION);

  const meanZ = centers.map((_, r) => {
    const members = rows.filter((_, i) => assign[i] === r);
    return MAP_MEASURES.map((_, j) => members.reduce((s, row) => s + row[j], 0) / members.length);
  });
  const words = nameRegions(meanZ);
  const genres = [...new Set(ids.map((id) => tagOf.get(id)!))].sort();
  const songs: Record<string, MapSong> = {};
  ids.forEach((id, i) => {
    songs[String(id)] = [r3(points[i][0]), r3(points[i][1]), assign[i], genres.indexOf(tagOf.get(id)!)];
  });

  const map: SoundMap = {
    v: 1,
    builtAt: new Date().toISOString().slice(0, 10),
    settings: SETTINGS,
    measures: [...MAP_MEASURES],
    mean: mean.map(r4),
    sd: sd.map(r4),
    genres,
    regions: centers.map((c, r) => ({ x: r3(c[0]), y: r3(c[1]), words: words[r], songs: assign.filter((a) => a === r).length })),
    songs,
  };
  fs.writeFileSync(path.join(ROOT, 'assets/sound-map.json'), JSON.stringify(map));
  console.log(`${ids.length} songs, ${map.regions.length} regions:`);
  for (const r of map.regions) console.log(`  ${r.words.padEnd(36)} ${r.songs}`);
}

main();

// lib/dark-map.ts
// Lighting the Dark Map: which songs you've heard and how brightly, how much of the map that lights, which lights
// are new since your last look, and each region's tally. Pure; lib/dark-map-app.ts feeds it from storage.

import type { SwipeEntry } from './discovery.ts';
import type { SoundFeatures } from './sound.ts';
import { measureVector, placeByNeighbors, standardize, type MapPoint, type Neighbor, type SoundMap } from './sound-map.ts';

/** 24 seconds, 80% of a preview: "fully heard", as for the piece's prints. */
export const FULL_LISTEN_MS = 24_000;
export const MIN_BRIGHTNESS = 0.25;
/** An entry from before listen times were recorded. */
export const UNKNOWN_BRIGHTNESS = 0.5;
/** The grid the lit share is counted on: 24 across, 36 down. */
export const GRID = { cols: 24, rows: 36 } as const;
/** The tap ring's radius, in map widths. */
export const LENS = 0.08;
/** Above this share of a ring heard, the sheet says "You know this part." */
export const KNOWN_SHARE = 0.8;

const HEARD = new Set<SwipeEntry['action']>(['skip', 'like', 'reveal', 'genre-jump']);

export type Light = { id: number; x: number; y: number; b: number; saved: boolean; fresh: boolean };
export type RegionStat = { region: number; words: string; songs: number; heard: number };
export type MapView = {
  /** Every map song. */
  dots: MapPoint[];
  /** Heard songs that have a place on the map. */
  lights: Light[];
  /** 0–1, on the GRID. */
  litShare: number;
  /** Heard songs with no place (no sound data), left out of the share. */
  unmapped: number;
  /** Least heard first, for the list. */
  regions: RegionStat[];
  /** Saved after the bloom, so the next look knows what's new. */
  litIds: number[];
};

/** Every song with a heard entry (skip, like, reveal, genre jump): the same rule as countSongsHeard. */
export function heardIds(history: SwipeEntry[]): Set<number> {
  return new Set(history.filter((e) => HEARD.has(e.action)).map((e) => e.trackId));
}

/** How brightly each heard song shines, 0.25–1: its longest listen, full at 24 seconds. Saved songs shine fully. */
export function brightness(history: SwipeEntry[], saved: Iterable<number>): Map<number, number> {
  const out = new Map<number, number>();
  for (const e of history) {
    if (!HEARD.has(e.action)) continue;
    const raw = e.listenMs == null ? UNKNOWN_BRIGHTNESS : e.listenMs / FULL_LISTEN_MS;
    const b = Number.isFinite(raw) ? Math.min(1, Math.max(MIN_BRIGHTNESS, raw)) : MIN_BRIGHTNESS;
    out.set(e.trackId, Math.max(out.get(e.trackId) ?? 0, b));
  }
  for (const id of saved) out.set(id, 1);
  return out;
}

/** A heard song's glow, in map widths. */
export const glowRadius = (b: number) => 0.035 * (0.6 + 0.4 * b);

const cellOf = (p: MapPoint) => `${Math.min(GRID.cols - 1, Math.floor(p.x * GRID.cols))},${Math.min(GRID.rows - 1, Math.floor(p.y * GRID.rows))}`;

/** The share of the grid cells holding a map song that also hold a heard one. */
export function litShare(all: MapPoint[], heard: MapPoint[]): number {
  const cells = new Set(all.map(cellOf));
  if (cells.size === 0) return 0;
  const lit = new Set(heard.map(cellOf).filter((c) => cells.has(c)));
  return lit.size / cells.size;
}

/** Lit songs that weren't lit at the last look: all of them on the first. */
export function newLights(lit: number[], seen: number[]): Set<number> {
  const before = new Set(seen);
  return new Set(lit.filter((id) => !before.has(id)));
}

/**
 * Everything the map shows. Heard songs on the map light their own spot; heard songs off it are placed by their
 * nearest neighbors when the sound cache measured them (`featuresOf`), and counted as unmapped otherwise.
 * `neighbors` is only called when a song needs placing.
 */
export function lightMap(
  map: SoundMap,
  history: SwipeEntry[],
  saved: number[],
  seen: number[],
  featuresOf: (id: number) => SoundFeatures | null,
  neighbors: () => Neighbor[]
): MapView {
  const savedSet = new Set(saved);
  const dots = Object.values(map.songs).map(([x, y]) => ({ x, y }));
  const regionHeard = map.regions.map(() => 0);
  const lights: Light[] = [];
  let unmapped = 0;
  for (const [id, b] of brightness(history, saved)) {
    const onMap = map.songs[String(id)];
    if (onMap) {
      regionHeard[onMap[2]]++;
      lights.push({ id, x: onMap[0], y: onMap[1], b, saved: savedSet.has(id), fresh: false });
      continue;
    }
    const f = featuresOf(id);
    const placed = f ? placeByNeighbors(standardize(map, measureVector(f)), neighbors()) : null;
    if (placed) lights.push({ id, x: placed.x, y: placed.y, b, saved: savedSet.has(id), fresh: false });
    else unmapped++;
  }
  const fresh = newLights(
    lights.map((l) => l.id),
    seen
  );
  for (const l of lights) l.fresh = fresh.has(l.id);
  const regions = map.regions
    .map((r, i) => ({ region: i, words: r.words, songs: r.songs, heard: regionHeard[i] }))
    .sort((a, b) => a.heard / a.songs - b.heard / b.songs || a.region - b.region);
  return { dots, lights, litShare: litShare(dots, lights), unmapped, regions, litIds: lights.map((l) => l.id) };
}

// components/map/map-geometry.ts
// Where the Dark Map's points sit on the canvas and on screen, zoomed about the canvas center. Worklets, so the
// gestures can use them on the UI thread. No Skia here: dark-map.tsx imports this without loading Skia.

import type { MapPoint } from '@/lib/sound-map';

/** px between the map's edge and its outermost songs. */
export const MAP_PAD = 14;
export const MAP_GROUND = '#0b1222';
export const MAP_DIM = '#3a4766';

/** A map point (0–1) on the unzoomed canvas, px. */
export function toCanvas(p: MapPoint, width: number, height: number): { x: number; y: number } {
  'worklet';
  return { x: MAP_PAD + p.x * (width - 2 * MAP_PAD), y: MAP_PAD + p.y * (height - 2 * MAP_PAD) };
}

/** Where a canvas point shows on screen at this zoom and pan. */
export function toScreen(c: { x: number; y: number }, width: number, height: number, zoom: number, panX: number, panY: number) {
  'worklet';
  return { x: width / 2 + panX + (c.x - width / 2) * zoom, y: height / 2 + panY + (c.y - height / 2) * zoom };
}

/** The map point under a screen point, or null outside the map. */
export function toMap(sx: number, sy: number, width: number, height: number, zoom: number, panX: number, panY: number): MapPoint | null {
  'worklet';
  const cx = (sx - width / 2 - panX) / zoom + width / 2;
  const cy = (sy - height / 2 - panY) / zoom + height / 2;
  const x = (cx - MAP_PAD) / (width - 2 * MAP_PAD);
  const y = (cy - MAP_PAD) / (height - 2 * MAP_PAD);
  return x < 0 || x > 1 || y < 0 || y > 1 ? null : { x, y };
}

/** Pan limits at a zoom, so the map never slides off screen. */
export function clampPan(v: number, zoom: number, size: number): number {
  'worklet';
  const m = (size * (zoom - 1)) / 2;
  return Math.min(m, Math.max(-m, v));
}

// components/map/dark-map-canvas.tsx
import { BlurMask, Canvas, Group, Path, Skia } from '@shopify/react-native-skia';
import { useMemo } from 'react';
import { useDerivedValue, type SharedValue } from 'react-native-reanimated';

import { Colors } from '@/constants/theme';
import { glowRadius, type Light } from '@/lib/dark-map';
import type { MapPoint } from '@/lib/sound-map';

import { MAP_DIM, MAP_PAD } from './map-geometry';

export type DarkMapCanvasProps = {
  width: number;
  height: number;
  dots: MapPoint[];
  lights: Light[];
  zoom: SharedValue<number>;
  panX: SharedValue<number>;
  panY: SharedValue<number>;
  /** 0 → 1 as the lights that are new since the last look come in. */
  bloom: SharedValue<number>;
};

/** The Dark Map: a dim dot for every song, a soft glow and a cream dot for each heard one, red for saves. Load it through dark-map.tsx. */
export default function DarkMapCanvas({ width, height, dots, lights, zoom, panX, panY, bloom }: DarkMapCanvasProps) {
  const layers = useMemo(() => {
    const sx = (p: MapPoint) => MAP_PAD + p.x * (width - 2 * MAP_PAD);
    const sy = (p: MapPoint) => MAP_PAD + p.y * (height - 2 * MAP_PAD);
    const circles = (list: MapPoint[], radius: (i: number) => number) => {
      const b = Skia.PathBuilder.Make();
      list.forEach((p, i) => b.addCircle(sx(p), sy(p), radius(i)));
      return b.detach();
    };
    const old = lights.filter((l) => !l.fresh);
    const fresh = lights.filter((l) => l.fresh);
    const glow = (list: Light[]) => circles(list, (i) => glowRadius(list[i].b) * (width - 2 * MAP_PAD));
    return {
      dim: circles(dots, () => 1.2),
      glowOld: glow(old),
      glowNew: glow(fresh),
      litOld: circles(old.filter((l) => !l.saved), () => 1.8),
      savedOld: circles(old.filter((l) => l.saved), () => 2.5),
      litNew: circles(fresh.filter((l) => !l.saved), () => 1.8),
      savedNew: circles(fresh.filter((l) => l.saved), () => 2.5),
      freshAt: fresh.map((l) => [sx(l), sy(l)]),
    };
  }, [dots, lights, width, height]);

  const transform = useDerivedValue(() => [
    { translateX: width / 2 + panX.get() },
    { translateY: height / 2 + panY.get() },
    { scale: zoom.get() },
    { translateX: -width / 2 },
    { translateY: -height / 2 },
  ]);
  const freshAt = layers.freshAt;
  // A ring opens around each new light as it blooms.
  const rings = useDerivedValue(() => {
    const b = Skia.PathBuilder.Make();
    const r = 3 + 14 * bloom.get();
    for (const [x, y] of freshAt) b.addCircle(x, y, r);
    return b.detach();
  });
  const ringOpacity = useDerivedValue(() => 0.8 * (1 - bloom.get()));
  const newGlowOpacity = useDerivedValue(() => 0.16 * bloom.get());
  const newOpacity = useDerivedValue(() => bloom.get());

  return (
    <Canvas style={{ width, height }} pointerEvents="none">
      <Group transform={transform}>
        <Path path={layers.glowOld} color={Colors.text} opacity={0.16}>
          <BlurMask blur={6} style="normal" />
        </Path>
        <Path path={layers.glowNew} color={Colors.text} opacity={newGlowOpacity}>
          <BlurMask blur={6} style="normal" />
        </Path>
        <Path path={layers.dim} color={MAP_DIM} opacity={0.6} />
        <Path path={layers.litOld} color={Colors.text} />
        <Path path={layers.savedOld} color={Colors.signal} />
        <Group opacity={newOpacity}>
          <Path path={layers.litNew} color={Colors.text} />
          <Path path={layers.savedNew} color={Colors.signal} />
        </Group>
        <Path path={rings} color={Colors.text} style="stroke" strokeWidth={1} opacity={ringOpacity} />
      </Group>
    </Canvas>
  );
}

// components/map/dark-map.tsx
import { useEffect } from 'react';
import { StyleSheet, View } from 'react-native';
import { Gesture, GestureDetector } from 'react-native-gesture-handler';
import Animated, { runOnJS, useAnimatedStyle, useSharedValue, withSpring, withTiming, type SharedValue } from 'react-native-reanimated';

import { lazySkia } from '@/components/print/lazy-skia';
import { ThemedText } from '@/components/themed-text';
import { Colors, Fonts } from '@/constants/theme';
import type { Light } from '@/lib/dark-map';
import type { MapPoint, MapRegion } from '@/lib/sound-map';

import type { DarkMapCanvasProps } from './dark-map-canvas';
import { clampPan, MAP_GROUND, MAP_PAD, toCanvas, toMap, toScreen } from './map-geometry';

const Canvas = lazySkia<DarkMapCanvasProps>(() => import('./dark-map-canvas'));
const MAX_ZOOM = 4;
const DOUBLE_TAP_ZOOM = 2.5;
const SPRING = { damping: 18, stiffness: 180 };
const LABEL_W = 180;

type Props = {
  width: number;
  height: number;
  dots: MapPoint[];
  lights: Light[];
  regions: (MapRegion & { lit: boolean })[];
  /** The tapped spot, ringed. */
  lens: MapPoint | null;
  /** In map widths. */
  lensRadius: number;
  onTap: (p: MapPoint) => void;
  /** New lights have bloomed in: time to remember this look. */
  onBloomDone: () => void;
  label: string;
  /** False for a still (You → Discoveries): no zoom, no taps. */
  interactive?: boolean;
};

/** The Dark Map with pinch zoom (1×–4×), pan, double-tap zoom and tap-to-look. The same gestures as the Tasteform. */
export function DarkMap({ width, height, dots, lights, regions, lens, lensRadius, onTap, onBloomDone, label, interactive = true }: Props) {
  const zoom = useSharedValue(1);
  const panX = useSharedValue(0);
  const panY = useSharedValue(0);
  const start = useSharedValue({ z: 1, x: 0, y: 0, fx: 0, fy: 0 });
  const bloom = useSharedValue(lights.some((l) => l.fresh) ? 0 : 1);

  // New lights bloom in once, when the map opens.
  useEffect(() => {
    if (bloom.get() >= 1) return;
    bloom.set(
      withTiming(1, { duration: 600 }, (done) => {
        if (done) runOnJS(onBloomDone)();
      })
    );
    // Once per mount.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const pinch = Gesture.Pinch()
    .enabled(interactive)
    .onStart((e) => {
      start.set({ z: zoom.get(), x: panX.get(), y: panY.get(), fx: e.focalX, fy: e.focalY });
    })
    .onUpdate((e) => {
      const s = start.get();
      const z = Math.min(MAX_ZOOM, Math.max(1, s.z * e.scale));
      // Keep the point that started under your fingers under them.
      const cx = (s.fx - width / 2 - s.x) / s.z;
      const cy = (s.fy - height / 2 - s.y) / s.z;
      zoom.set(z);
      panX.set(clampPan(e.focalX - width / 2 - cx * z, z, width));
      panY.set(clampPan(e.focalY - height / 2 - cy * z, z, height));
    })
    .onEnd(() => {
      if (zoom.get() < 1.08) {
        zoom.set(withSpring(1, SPRING));
        panX.set(withSpring(0, SPRING));
        panY.set(withSpring(0, SPRING));
      }
    });

  const pan = Gesture.Pan()
    .enabled(interactive)
    .manualActivation(true)
    .onTouchesMove((_e, manager) => {
      if (zoom.get() > 1.02) manager.activate();
      else manager.fail();
    })
    .onStart(() => {
      start.set({ ...start.get(), x: panX.get(), y: panY.get() });
    })
    .onUpdate((e) => {
      panX.set(clampPan(start.get().x + e.translationX, zoom.get(), width));
      panY.set(clampPan(start.get().y + e.translationY, zoom.get(), height));
    });

  const doubleTap = Gesture.Tap()
    .enabled(interactive)
    .numberOfTaps(2)
    .maxDelay(250)
    .onEnd((e, ok) => {
      if (!ok) return;
      if (zoom.get() > 1.08) {
        zoom.set(withSpring(1, SPRING));
        panX.set(withSpring(0, SPRING));
        panY.set(withSpring(0, SPRING));
      } else {
        const z = DOUBLE_TAP_ZOOM;
        zoom.set(withSpring(z, SPRING));
        panX.set(withSpring(clampPan((e.x - width / 2) * (1 - z), z, width), SPRING));
        panY.set(withSpring(clampPan((e.y - height / 2) * (1 - z), z, height), SPRING));
      }
    });

  const tap = Gesture.Tap()
    .enabled(interactive)
    .maxDuration(300)
    .maxDistance(12)
    .onEnd((e, ok) => {
      if (!ok) return;
      const p = toMap(e.x, e.y, width, height, zoom.get(), panX.get(), panY.get());
      if (p) runOnJS(onTap)(p);
    });

  const gesture = Gesture.Simultaneous(pinch, pan, Gesture.Exclusive(doubleTap, tap));

  return (
    <GestureDetector gesture={gesture}>
      <View style={[styles.map, { width, height }]} accessible accessibilityLabel={label}>
        <Canvas width={width} height={height} dots={dots} lights={lights} zoom={zoom} panX={panX} panY={panY} bloom={bloom} />
        {regions.map((r, i) => (
          <MapLabel key={i} region={r} width={width} height={height} zoom={zoom} panX={panX} panY={panY} />
        ))}
        {lens && <MapLens point={lens} radius={lensRadius} width={width} height={height} zoom={zoom} panX={panX} panY={panY} />}
      </View>
    </GestureDetector>
  );
}

type Motion = { width: number; height: number; zoom: SharedValue<number>; panX: SharedValue<number>; panY: SharedValue<number> };

/** A region's words, at its center, the same size at any zoom. Cream where you've heard some of it. */
function MapLabel({ region, width, height, zoom, panX, panY }: Motion & { region: MapRegion & { lit: boolean } }) {
  const at = toCanvas(region, width, height);
  const style = useAnimatedStyle(() => {
    const s = toScreen(at, width, height, zoom.get(), panX.get(), panY.get());
    return { transform: [{ translateX: s.x - LABEL_W / 2 }, { translateY: s.y - 7 }] };
  });
  return (
    <Animated.View pointerEvents="none" style={[styles.label, style]}>
      <ThemedText style={[styles.labelText, { color: region.lit ? Colors.text : Colors.textTertiary }]} numberOfLines={1}>
        {region.words.toUpperCase()}
      </ThemedText>
    </Animated.View>
  );
}

/** The dashed ring around a tapped spot. */
function MapLens({ point, radius, width, height, zoom, panX, panY }: Motion & { point: MapPoint; radius: number }) {
  const at = toCanvas(point, width, height);
  const r = radius * (width - 2 * MAP_PAD);
  const style = useAnimatedStyle(() => {
    const s = toScreen(at, width, height, zoom.get(), panX.get(), panY.get());
    const rr = r * zoom.get();
    return { width: 2 * rr, height: 2 * rr, borderRadius: rr, transform: [{ translateX: s.x - rr }, { translateY: s.y - rr }] };
  });
  return <Animated.View pointerEvents="none" style={[styles.lens, style]} />;
}

const styles = StyleSheet.create({
  map: { backgroundColor: MAP_GROUND, overflow: 'hidden' },
  label: { position: 'absolute', left: 0, top: 0, width: LABEL_W, alignItems: 'center' },
  labelText: {
    fontFamily: Fonts.mono,
    fontSize: 9,
    lineHeight: 14,
    letterSpacing: 0.8,
    textShadowColor: MAP_GROUND,
    textShadowRadius: 3,
  },
  lens: {
    position: 'absolute',
    left: 0,
    top: 0,
    borderWidth: 1.2,
    borderStyle: 'dashed',
    borderColor: Colors.text,
    backgroundColor: 'rgba(243, 234, 216, 0.06)',
  },
});

```tsx
// app/map.tsx
import { Ionicons } from '@expo/vector-icons';
import * as Haptics from 'expo-haptics';
import { useRouter } from 'expo-router';
import { useEffect, useMemo, useState } from 'react';
import { type LayoutChangeEvent, Pressable, StyleSheet, View } from 'react-native';
import { useSafeAreaInsets } from 'react-native-safe-area-context';

import { DarkMap } from '@/components/map/dark-map';
import { MAP_DIM, MAP_GROUND } from '@/components/map/map-geometry';
import { RegionList } from '@/components/map/region-list';
import { SpotSheet } from '@/components/map/spot-sheet';
import { ThemedText } from '@/components/themed-text';
import { Colors, Fonts, Spacing, Ui } from '@/constants/theme';
import { LENS, type MapView } from '@/lib/dark-map';
import { loadMapView } from '@/lib/dark-map-app';
import { saveMapSeen } from '@/lib/discovery-storage';
import { loadSoundMap, regionAt, songsWithin, type MapPoint } from '@/lib/sound-map';
import { requestSteer } from '@/lib/steer-request';

type Spot = { point: MapPoint; words: string; songs: number; heard: number };

/** The Dark Map: every measured song placed by how it sounds. What you've heard glows; the dark is your blind spot. */
export default function MapScreen() {
  const router = useRouter();
  const insets = useSafeAreaInsets();
  const map = loadSoundMap();
  const [view, setView] = useState<MapView | null>(null);
  const [area, setArea] = useState<{ width: number; height: number } | null>(null);
  const [spot, setSpot] = useState<Spot | null>(null);
  const [listOpen, setListOpen] = useState(false);

  useEffect(() => {
    loadMapView().then(setView);
  }, []);

  const lit = useMemo(() => new Set(view?.litIds ?? []), [view]);
  // A region's words are cream once you've heard a little of it (6%), dim before.
  const labels = useMemo(
    () => (map && view ? map.regions.map((r, i) => ({ ...r, lit: (view.regions.find((s) => s.region === i)?.heard ?? 0) > 0.06 * r.songs })) : []),
    [map, view]
  );
  const pct = view ? Math.round(view.litShare * 100) : 0;

  function look(point: MapPoint) {
    if (!map) return;
    Haptics.selectionAsync();
    const near = songsWithin(map, point, LENS);
    setSpot({ point, words: map.regions[regionAt(map, point, LENS)].words, songs: near.length, heard: near.filter((id) => lit.has(id)).length });
  }

  function lookAtRegion(region: number) {
    if (!map) return;
    setListOpen(false);
    look({ x: map.regions[region].x, y: map.regions[region].y });
  }

  // Home takes the request when it's focused again, the same way the Liked list steers it.
  function go() {
    if (!spot || spot.songs === 0) return;
    requestSteer({ kind: 'map', point: spot.point, words: spot.words });
    router.back();
    router.navigate('/');
  }

  function onArea(e: LayoutChangeEvent) {
    const { width, height } = e.nativeEvent.layout;
    setArea((a) => (a && a.width === width && a.height === height ? a : { width, height }));
  }

  return (
    <View style={[styles.screen, { paddingTop: insets.top + Spacing.sm, paddingBottom: insets.bottom + Spacing.sm }]}>
      <View style={styles.header}>
        <Pressable onPress={() => router.back()} hitSlop={10} accessibilityLabel="Close" style={Ui.textButton}>
          <Ionicons name="close" size={26} color={Colors.textSecondary} />
        </Pressable>
        <ThemedText style={styles.mono}>DARK MAP</ThemedText>
        <View style={styles.headerRight}>
          <ThemedText style={styles.mono} accessibilityLabel={`${pct} percent lit`}>
            {pct}% LIT
          </ThemedText>
          <Pressable
            onPress={() => setListOpen((v) => !v)}
            hitSlop={10}
            style={Ui.textButton}
            accessibilityRole="button"
            accessibilityLabel={listOpen ? 'Show the map' : 'Show the regions as a list'}>
            <ThemedText style={[styles.mono, styles.toggle]}>{listOpen ? 'MAP' : 'LIST'}</ThemedText>
          </Pressable>
        </View>
      </View>

      <View style={styles.area} onLayout={onArea}>
        {!map ? (
          <ThemedText style={styles.dim}>The map isn&apos;t built yet.</ThemedText>
        ) : listOpen && view ? (
          <RegionList regions={view.regions} onPick={lookAtRegion} />
        ) : area && view ? (
          <DarkMap
            width={area.width}
            height={area.height}
            dots={view.dots}
            lights={view.lights}
            regions={labels}
            lens={spot?.point ?? null}
            lensRadius={LENS}
            onTap={look}
            onBloomDone={() => saveMapSeen(view.litIds)}
            label={`Dark map, ${pct} percent lit. Use List to explore the regions.`}
          />
        ) : null}
      </View>

      <View style={styles.legend}>
        <ThemedText style={styles.key}>
          <ThemedText style={[styles.key, { color: Colors.text }]}>● </ThemedText>HEARD
        </ThemedText>
        <ThemedText style={styles.key}>
          <ThemedText style={[styles.key, { color: Colors.signal }]}>● </ThemedText>SAVED
        </ThemedText>
        <ThemedText style={styles.key}>
          <ThemedText style={[styles.key, { color: MAP_DIM }]}>● </ThemedText>NOT YET
        </ThemedText>
      </View>
      <ThemedText style={styles.hint}>Tap anywhere dark to go there.</ThemedText>
      {!!view && view.unmapped > 0 && (
        <ThemedText style={styles.hint}>
          {view.unmapped === 1 ? "1 song you heard isn't on the map yet." : `${view.unmapped} songs you heard aren't on the map yet.`}
        </ThemedText>
      )}

      {spot && <SpotSheet words={spot.words} songs={spot.songs} heard={spot.heard} bottom={insets.bottom} onGo={go} onClose={() => setSpot(null)} />}
    </View>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, backgroundColor: Colors.background },
  header: { flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between', paddingHorizontal: Spacing.lg },
  headerRight: { flexDirection: 'row', alignItems: 'center', gap: Spacing.md },
  mono: { fontFamily: Fonts.mono, fontSize: 12, letterSpacing: 1.2, color: Colors.text },
  toggle: { textDecorationLine: 'underline' },
  area: { flex: 1, backgroundColor: MAP_GROUND, justifyContent: 'center' },
  dim: { textAlign: 'center', color: Colors.textSecondary },
  legend: { flexDirection: 'row', justifyContent: 'center', gap: Spacing.lg, paddingTop: Spacing.md },
  key: { fontFamily: Fonts.mono, fontSize: 10, letterSpacing: 1, color: Colors.textSecondary },
  hint: { textAlign: 'center', fontSize: 13, color: Colors.textSecondary, paddingTop: Spacing.xs },
});
```
